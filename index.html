import express from "express"; import cors from "cors"; import Stripe from "stripe"; const app = express(); app.use(cors()); app.use(express.json()); const stripe = new Stripe(process.env.STRIPE_SECRET_KEY); app.get("/", (req, res) => { res.send("Backend Stripe opérationnel !"); }); app.post("/pay", async (req, res) => { try { const session = await stripe.checkout.sessions.create({ mode: "payment", payment_method_types: ["card"], line_items: [ { price_data: { currency: "eur", product_data: { name: "Abonnement Jamess Conseil" }, unit_amount: 1000 }, quantity: 1 } ], success_url: "https://jamess-conseil.com", cancel_url: "https://jamess-conseil.com" }); res.json({ url: session.url }); } catch (e) { res.status(500).json({ error: e.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Serveur lancé sur le port ${PORT}`); });
