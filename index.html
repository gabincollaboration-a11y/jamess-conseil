import express from "express"; import Stripe from "stripe"; import cors from "cors"; const app = express(); app.use(cors()); app.use(express.json()); const stripe = new Stripe(process.env.STRIPE_SECRET_KEY); app.post("/pay", async (req, res) => { try { const session = await stripe.checkout.sessions.create({ payment_method_types: ["card"], mode: "payment", line_items: [ { price_data: { currency: "eur", product_data: { name: "Produit test", }, unit_amount: 1000, }, quantity: 1, }, ], success_url: "https://ton-site.com/success", cancel_url: "https://ton-site.com/cancel", }); res.json({ url: session.url }); } catch (err) { res.status(500).json({ error: err.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log("Server OK"));
